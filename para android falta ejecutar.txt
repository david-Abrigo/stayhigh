### CONTEXTO DE BASE DE DATOS (Supabase)
En Supabase, la tabla `precharges` contiene:
- `id` (uuid): ID primario.
- `public_id` (text): Código único del cobro (ej. `PRE-123456`).
- `expected_name` (text): Nombre del cliente esperado / registrado.
- `expected_amount` (numeric): Monto cobrado (ej. 25.00).
- `currency` (text): Moneda (`PEN`).
- `description` (text, nullable): **EL MENSAJE O NOTA DEL COMPRADOR** (ej. "Mesa 4", "Sin cebolla", "Pedido #102").
- `status` (text): 'PENDING', 'MATCHED', 'EXPIRED'.
- `created_at` (timestamptz): Fecha de creación.
- `matched_at` (timestamptz, nullable): Fecha y hora exacta en que se confirmó el pago por Yape.
- `device_id` (text/uuid): ID del dispositivo del vendedor.
- `metadata` (jsonb): Datos extra, incluyendo `buyer_note`.

### PASO 1: Clase `PaymentLinkSecurity.kt` (Generador de links ultra-cortos)
Crea o actualiza `PaymentLinkSecurity.kt` en el paquete de seguridad/utilidades:

```kotlin
package com.example.security

import android.util.Base64
import org.json.JSONObject
import java.security.MessageDigest
import java.util.Locale

object PaymentLinkSecurity {
    private const val PAYMENT_SALT = "stayhigh_secure_salt_2026"
    private const val BASE_URL = "https://stayhigh.onrender.com"

    private fun computeSignature(deviceId: String, amount: Double): String {
        val formattedAmount = String.format(Locale.US, "%.2f", amount)
        val data = "$deviceId:$formattedAmount:$PAYMENT_SALT"
        val md = MessageDigest.getInstance("SHA-256")
        val digest = md.digest(data.toByteArray(Charsets.UTF_8))
        return digest.joinToString("") { "%02x".format(it) }.substring(0, 12)
    }

    /**
     * OPCIÓN 1: Enlace ultra-corto con monto prefijado
     * val link = PaymentLinkSecurity.generateFixedAmountLink(myDeviceId, 25.0)
     */
    fun generateFixedAmountLink(
        deviceId: String,
        amount: Double
    ): String {
        val sig = computeSignature(deviceId, amount)
        val json = JSONObject().apply {
            put("d", deviceId)
            put("a", amount)
            put("s", sig)
        }
        val token = Base64.encodeToString(
            json.toString().toByteArray(Charsets.UTF_8),
            Base64.URL_SAFE or Base64.NO_WRAP or Base64.NO_PADDING
        )
        return "$BASE_URL/?c=$token"
    }

    /**
     * OPCIÓN 2: Enlace ultra-corto con monto libre (el cliente digita el monto en la web)
     * val link = PaymentLinkSecurity.generateFreeAmountLink(myDeviceId)
     */
    fun generateFreeAmountLink(deviceId: String): String {
        val json = JSONObject().apply {
            put("d", deviceId)
        }
        val token = Base64.encodeToString(
            json.toString().toByteArray(Charsets.UTF_8),
            Base64.URL_SAFE or Base64.NO_WRAP or Base64.NO_PADDING
        )
        return "$BASE_URL/?c=$token"
    }
}
```

### PASO 3: Repositorio para consultar compras confirmadas (`ConfirmedPurchasesRepository.kt`)
Crea `ConfirmedPurchasesRepository.kt` para obtener las compras con status = MATCHED:

```kotlin
package com.example.data.repository

import com.example.data.supabase.ConfirmedPrechargeDto
import com.example.data.supabase.SupabaseClient
import io.github.jan.supabase.postgrest.postgrest
import io.github.jan.supabase.postgrest.query.Order
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext

class ConfirmedPurchasesRepository {

    /**
     * Obtiene las precharges en estado MATCHED para el dispositivo actual,
     * ordenadas por fecha de confirmación descendente.
     */
    suspend fun getConfirmedPurchases(deviceId: String): Result<List<ConfirmedPrechargeDto>> = withContext(Dispatchers.IO) {
        try {
            val result = SupabaseClient.client.postgrest["precharges"]
                .select {
                    filter {
                        eq("device_id", deviceId)
                        eq("status", "MATCHED")
                    }
                    order(column = "matched_at", order = Order.DESCENDING)
                    limit(50)
                }
                .decodeList<ConfirmedPrechargeDto>()
            Result.success(result)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}
```

```kotlin
package com.example.ui.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.data.repository.ConfirmedPurchasesRepository
import com.example.data.supabase.ConfirmedPrechargeDto
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

data class ConfirmedPurchasesUiState(
    val isLoading: Boolean = false,
    val purchases: List<ConfirmedPrechargeDto> = emptyList(),
    val errorMessage: String? = null,
    val searchQuery: String = ""
)

class ConfirmedPurchasesViewModel(
    private val repository: ConfirmedPurchasesRepository = ConfirmedPurchasesRepository()
) : ViewModel() {

    private val _uiState = MutableStateFlow(ConfirmedPurchasesUiState())
    val uiState: StateFlow<ConfirmedPurchasesUiState> = _uiState.asStateFlow()

    fun loadConfirmedPurchases(deviceId: String) {
        viewModelScope.launch {
            _uiState.value = _uiState.value.copy(isLoading = true, errorMessage = null)
            val result = repository.getConfirmedPurchases(deviceId)
            result.onSuccess { list ->
                _uiState.value = _uiState.value.copy(
                    isLoading = false,
                    purchases = list,
                    errorMessage = null
                )
            }.onFailure { error ->
                _uiState.value = _uiState.value.copy(
                    isLoading = false,
                    errorMessage = error.localizedMessage ?: "Error al cargar compras confirmadas"
                )
            }
        }
    }

    fun onSearchQueryChanged(query: String) {
        _uiState.value = _uiState.value.copy(searchQuery = query)
    }
}
```

```kotlin
private fun formatDateTime(isoString: String?): String {
    if (isoString.isNullOrBlank()) return ""
    return try {
        val parser = SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss", Locale.getDefault())
        val date = parser.parse(isoString) ?: return isoString
        val formatter = SimpleDateFormat("dd/MM/yyyy HH:mm", Locale.getDefault())
        formatter.format(date)
    } catch (e: Exception) {
        isoString.take(19).replace("T", " ")
    }
}
```
